cmake_minimum_required(VERSION 3.20)
project(rs3_native VERSION 1.0.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "rs3_native requires Windows x64")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile_generated.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_profile.py"
            "${CMAKE_CURRENT_BINARY_DIR}/profile_generated.h"
    DEPENDS generate_profile.py ../rs3_reader/profile_951_3.json VERBATIM)
add_library(rs3_native SHARED src/rs3_native.c "${CMAKE_CURRENT_BINARY_DIR}/profile_generated.h")
target_include_directories(rs3_native PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(rs3_native PRIVATE bcrypt)
if(MSVC)
    target_compile_definitions(rs3_native PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(rs3_native PRIVATE /W4)
else()
    target_compile_options(rs3_native PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_native tests/test_native.c "${CMAKE_CURRENT_BINARY_DIR}/profile_generated.h")
    target_include_directories(test_native PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(test_native PRIVATE bcrypt)
    target_compile_definitions(test_native PRIVATE _CRT_SECURE_NO_WARNINGS)
    add_test(NAME native COMMAND test_native)
endif()
add_executable(read_example examples/read.c)
target_include_directories(read_example PRIVATE include)
install(TARGETS rs3_native RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/rs3/rs3.h DESTINATION include/rs3)
set(RS3_CACHE_ROOT "C:/ProgramData/Jagex/RuneScape" CACHE PATH "Installed read-only game cache")
if(EXISTS "${RS3_CACHE_ROOT}/js5-19.jcache")
    add_custom_command(TARGET rs3_native POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_catalog.py"
                --cache-root "${RS3_CACHE_ROOT}" --output "$<TARGET_FILE_DIR:rs3_native>/rs3_catalog.bin" VERBATIM)
endif()
